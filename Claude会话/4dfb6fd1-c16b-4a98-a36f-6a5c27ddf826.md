```python
import os
import csv
from PIL import Image
from collections import defaultdict

DATA_DIR = r"D:\blip2-main\data"
IMAGE_DIR = os.path.join(DATA_DIR, "Images")
CAPTION_FILE = os.path.join(DATA_DIR, "captions.txt")


def load_flickr8k(limit=200):
    """加载前 limit 张唯一图片及其所有 captions。

    Returns:
        data: list[dict], 每个元素为 {"image": PIL.Image, "captions": [str, ...], "filename": str}
    """
    # 1. 读取 caption 文件，按图片名分组
    image_captions = defaultdict(list)
    with open(CAPTION_FILE, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            image_captions[row["image"]].append(row["caption"].strip())

    # 2. 取前 limit 张唯一图片
    unique_images = list(image_captions.keys())[:limit]

    # 3. 加载图片
    data = []
    for filename in unique_images:
        path = os.path.join(IMAGE_DIR, filename)
        img = Image.open(path).convert("RGB")
        data.append({
            "image": img,
            "captions": image_captions[filename],
            "filename": filename,
        })

    return data


if __name__ == "__main__":
    data = load_flickr8k(limit=200)
    print(f"已加载 {len(data)} 张图片")
    for i, item in enumerate(data[:3]):
        print(f"\n[{i}] {item['filename']}  ({item['image'].size})")
        for cap in item["captions"]:
            print(f"    - {cap}")

```

**使用方式：**

```python
from load_flickr8k import load_flickr8k

data = load_flickr8k(limit=200)

# data 是一个 list，每个元素包含：
#   data[i]["image"]      — PIL.Image 对象
#   data[i]["captions"]   — 该图片对应的 5 条 caption 列表
#   data[i]["filename"]   — 图片文件名
```

**代码逻辑：**

1. 用 `csv.DictReader` 解析 `captions.txt`，以 `image` 字段为 key 分组所有 caption
2. 取前 200 张唯一图片名
3. 用 PIL 加载每张图片，连同其所有 caption 一起返回

总共会加载 200 张图片，每张对应 5 条 caption。
